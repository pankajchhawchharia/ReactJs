111. break array into smaller subarrays of specific size - array chunking- display data in pages or grids, processing large data in smaller batches
Sol: function chunking(arr,size) {
    let result =[];
    for(let i=0;i<arr.length;i = i+size) {
        result.push(arr.slice(i,i+size));
    }
    return result;
}
const arr = [1,2,3,4,5,6,7,8,9];
const chunked = chunking(arr,3);
console.log(chunked); // [ [ 1, 2, 3 ], [ 4, 5, 6 ], [ 7, 8, 9 ] ]

106. custom hook to write eventlistner for resize
Sol: import { useState, useEffect } from 'react';
function useWindowSize() {
  const [size, setSize] = useState({
    width: window.innerWidth,
    height: window.innerHeight,
  });
  useEffect(() => {
    const handleResize = () => {
      setSize({
        width: window.innerWidth,
        height: window.innerHeight,
      });
    };
    window.addEventListener('resize', handleResize);
    handleResize();
    return () => window.removeEventListener('resize', handleResize);
  }, []);
  return size;
}
export default useWindowResize;

sol: let arr1 = [1,4,9];
let arr2 = [2,8,6];
// merge
function mergee(arr1,arr2) {
    let newArr = [];
    for(let i=0;i<arr1.length;i++) {
        newArr.push(arr1[i]);  
    }
    for (let i =0 ;i<arr2.length;i++) {
        newArr.push(arr2[i]);
    }
    // bubble sorting
    for(let i=0;i<newArr.length-1;i++) {
        for (let j=0;j<newArr.length-i-1;j++) {
            if(newArr[j] > newArr[j+1] ) {
              let temp = newArr[j];
              newArr[j] = newArr[j+1];
              newArr[j+1] = temp;
            }
        }
    }
    return newArr;
}
console.log(mergee(arr1,arr2)); // [1,2,4,6,8,9]
