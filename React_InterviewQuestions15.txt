import React, { useState } from 'react';
function Child({ onSubmit }) {
  const [name, setName] = useState('');
  const handleSubmit = () => {
    onSubmit({ name });
  };
  return (
    <div>
      <p>hi from child </p>
      <input onChange={(e) => setName(e.target.value)} />
      <button onClick={handleSubmit}>click</button>
    </div>
  );
}
export default Child;
