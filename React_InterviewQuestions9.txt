34. Write a code in React to change the background on hover

import React, { useState } from 'react';
const DynamicHover = () => {
  const [isHovered, setIsHovered] = useState(false);
  const style = {
    width: '200px',
    height: '200px',
    backgroundColor: isHovered ? '#e74c3c' : '#2ecc71',
    transition: 'background-color 0.3s ease',
    display: 'flex',
    alignItems: 'center',
    justifyContent: 'center',
    cursor: 'pointer'
  };
  return (
    <div 
      style={style}
      onMouseEnter={() => setIsHovered(true)} 
      onMouseLeave={() => setIsHovered(false)}
    >
      {isHovered ? 'Active!' : 'Hover Me'}
    </div>
  );
};
export default DynamicHover;

import { render, screen, fireEvent } from '@testing-library/react';
import { Provider } from 'react-redux';
import { configureStore } from '@reduxjs/toolkit';
import rootReducer from './reducer'; // Your actual reducer
import MyComponent from './MyComponent';

test('button click updates redux store state', () => {
  // 1. Create a fresh store instance for the test
  const store = configureStore({ reducer: rootReducer });

  // 2. Assert initial state (Before Click)
  expect(store.getState().counter.value).toBe(0);

  // Render component wrapped in Provider
  render(
    <Provider store={store}>
      <MyComponent />
    </Provider>
  );

  // 3. Simulate button click
  const button = screen.getByRole('button', { name: /increment/i });
  fireEvent.click(button);

  // 4. Assert updated state (After Click)
  expect(store.getState().counter.value).toBe(1);
});
