const Checkbox = () => {
  const [selectedBranches, setSelectedBranches] = useState({});

  const handleParent = (cluster) => {
    const newState = { ...selectedBranches };
    const isAllSelected = cluster.branches.every(b => newState[b.name]);
    
    // Toggle all branches in this cluster
    cluster.branches.forEach(branch => {
      newState[branch.name] = !isAllSelected;
    });
    setSelectedBranches(newState);
  };

  const handleChild = (name) => {
    setSelectedBranches(prev => ({ ...prev, [name]: !prev[name] }));
  };

  return (
    <div>
      {data.map((cluster) => (
        <div key={cluster.name}>
          <h3>
            {cluster.name} 
            <input type="checkbox" onChange={() => handleParent(cluster)} />
          </h3>
          {cluster.branches.map(branch => (
            <div key={branch.name} className="ml-4">
              {branch.name}
              <input 
                type="checkbox" 
                checked={!!selectedBranches[branch.name]} 
                onChange={() => handleChild(branch.name)} 
              />
            </div>
          ))}
        </div>
      ))}
    </div>
  );
};

4. How do you implement the debounce function correctly in React to prevent a search function from firing on every keystroke?
import { useCallback } from "react";

// Standard debounce utility
function debounce(fn, delay) {
  let timer;
  return function (...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), delay);
  };
}

// Inside a component:
const handleSearch = (e) => console.log(e.target.value);
