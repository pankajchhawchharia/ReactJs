4. Identify a redundant keyword in the return statement of this asyncTask function.
// async function asyncTask() {
    try {
        const valueA = await functionA();
        const valueB = await functionB(valueA);
        const valueC = await functionC(valueB);
        return await functionD(valueC);
    } catch (err) {
        logger.error(err);
    }
}
The await in return await is redundant. Since the function is already marked async, it will automatically wrap the returned value (even a promise from functionD) in a new promise.

Correction: Use return functionD(valueC);.

6. Write a function to group the input array ["eat", "ate", "tea", "ant", "tan", "bat"] into anagram groups.
function groupAnagrams(strs) {
    const map = {};

    for (let str of strs) {
        // Sort string to create a unique key: "eat" -> "aet"
        let key = str.split('').sort().join('');
        
        if (!map[key]) map[key] = [];
        map[key].push(str);
    }

    return Object.values(map);
}

const input = ["eat", "ate", "tea", "ant", "tan", "bat"];
console.log(groupAnagrams(input));
// Output: [ ["eat", "ate", "tea"], ["ant", "tan"], ["bat"] ]
