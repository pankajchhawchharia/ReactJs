Sol: function countWordFrequency(str) {
  const wordCounts = {};
  const words = str.trim().split(/\s+/);
  for (const word of words) {
    const normalizedWord = word.toLowerCase();
    wordCounts[normalizedWord] = (wordCounts[normalizedWord] || 0) + 1;
  }
  return wordCounts;
}
