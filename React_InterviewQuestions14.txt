var groups = [
  { name: "Devi", age: '20' },
  { name: "Ravi", age: "30" },
  { name: "Ram", age: '20' }
];

 
//Output
{
  "20": [
    { "name": "Devi", "age": "20" },
    { "name": "Ram", "age": "20" }
  ],
  "30": [
    { "name": "Ravi", "age": "30" }
  ]
}

var groupedByAge = {};

groups.forEach(item => {
  if (!groupedByAge[item.age]) {
    groupedByAge[item.age] = [];
  }
  groupedByAge[item.age].push(item);
});

console.log(groupedByAge);

79. Reversing string without the inbuild methods (.split('').reverse().join('')) part 1 - 343 or part 2 - 36

function reverseStr(str) {
  let newStr = '';
  newStr = str.split('');
  let rev = '';
  // return str.split('').reverse().join('');
  for(let i = newStr.length - 1;i>=0;i--) {
    rev = rev + newStr[i];
  }
  return rev;
}
console.log(reverseStr("hello"));

80. Input string. output each character occurrences - part 2 - 165

function charCount(str) {
  let obj = {};
  for(let i = 0;i<str.length;i++) {
    char = str[i];
    obj[char] = (obj[char] || 0) + 1;
  }
  return obj;
}
console.log(charCount("hello")); 
// Output: { h: 1, e: 1, l: 2, o: 1 }
